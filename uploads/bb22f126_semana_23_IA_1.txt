import json

# --- 1. SUPER GERADOR DINÂMICO DE CONHECIMENTO (+18.000 REGISTROS) ---
# Lista massiva de tópicos abrangendo todo o universo da tecnologia e desenvolvimento
topicos_tecnologia = [
    "Python", "Banco de Dados", "API", "Variáveis", "Estruturas de repetição", 
    "Estruturas condicionais", "Funções", "HTML", "CSS", "JavaScript", 
    "Machine Learning", "Redes de Computadores", "Segurança da Informação", 
    "Lógica de Programação", "Git e GitHub", "Linux", "Docker", "Kubernetes",
    "SQL Server", "MySQL", "PostgreSQL", "MongoDB", "Firebase", "Node.js", 
    "React", "Angular", "Vue.js", "TypeScript", "Java", "C#", "C++", "Ruby", 
    "PHP", "Swift", "Kotlin", "Flutter", "React Native", "DevOps", "Cloud Computing", 
    "AWS", "Google Cloud", "Azure", "Inteligência Artificial", "Deep Learning", 
    "Processamento de Linguagem Natural", "Visão Computacional", "Ciência de Dados", 
    "Análise de Dados", "Excel Avançado", "Power BI", "Metodologias Ágeis", "Scrum", 
    "Kanban", "Engenharia de Software", "Arquitetura de Sistemas", "Microserviços", 
    "Testes de Software", "QA", "Cybersecurity", "Criptografia"
]

niveis_ensino = ["Iniciante", "Intermediário", "Avançado"]

documentos = []

# Três loops alinhados que multiplicam: 60 tópicos x 3 níveis x 100 módulos de aula = 18.000 combinações!
for topico in topicos_tecnologia:
    for nivel in niveis_ensino:
        for aula_id in range(1, 101):
            documentos.append({
                "titulo": f"{topico} {nivel} - Aula {aula_id}",
                "conteudo": f"Material Didático EstudaMais: Este documento contém a cobertura teórica completa e os exercícios práticos sobre '{topico}', focado especificamente no nível '{nivel}', correspondente ao planejamento pedagógico da Aula de número {aula_id}."
            })

# --- 2. SISTEMA DE RECUPERAÇÃO DE CONTEXTO OTIMIZADO ---
def recuperar_documento(pergunta):
    pergunta = pergunta.lower()
    
    # Varredura ultra rápida na memória do computador para localizar o material correto
    for doc in documentos:
        if doc["titulo"].lower() in pergunta:
            return doc["conteudo"]
            
    # BUSCA FLEXÍVEL: Se o aluno não digitou o nível ou a aula, busca apenas pelo nome do assunto principal
    for doc in documentos:
        # Se o aluno digitou apenas o nome do tópico (ex: "Docker"), traz a primeira aula disponível
        palavras_topico = doc["titulo"].split()
        if palavras_topico[0].lower() in pergunta:
            return f"[Busca Rápida] Encontrei algo relacionado ao seu tema! Veja o plano de {doc['titulo']}: {doc['conteudo']}"
            
    return None

# --- 3. CONTROLE DO HISTÓRICO DE CONVERSA ---
if 'historico_conversa' not in locals():
    historico_conversa = []

# --- 4. ENGINE PRINCIPAL DO AGENTE INTEGRADO ---
class AgenteDuvidas:
    def __init__(self, nome):
        self.nome = nome

    def responder(self, pergunta):
        pergunta_limpa = pergunta.lower().strip()
        
        # Registra a pergunta na memória local e cria o arquivo JSON persistente no Colab
        historico_conversa.append(pergunta)
        with open("memoria.json", "w") as arquivo:
            json.dump(historico_conversa, arquivo)
            
        # Busca no motor RAG de 18 mil registros
        contexto = recuperar_documento(pergunta_limpa)
        if contexto:
            return contexto
            
        # Respostas Rápidas / Camada de diálogo humano
        if pergunta_limpa in ["oi", "olá", "ola", "bom dia", "boa tarde", "boa noite", "eae", "salve"]:
            return f"Olá! Eu sou o {self.nome}. Minha base de dados foi expandida e agora conta com mais de 18.000 respostas e apostilas digitais. O que vamos estudar hoje?"
            
        elif pergunta_limpa in ["obrigado", "obrigada", "valeu", "tamo junto"]:
            return "Disponha! Estou aqui para acelerar o seu aprendizado na EstudaMais."
            
        elif "tamanho da base" in pergunta_limpa or "quantas respostas" in pergunta_limpa:
            return f"Impressionante, não é? Minha memória atual possui exatamente {len(documentos)} materiais e tópicos estruturados de tecnologia cadastrados."

        # Mensagem Dinâmica instruindo o formato ideal de entrada
        return "Não consegui mapear esse termo exato. Como nossa base é gigantesca, experimente pesquisar no formato: '[Tecnologia] [Nível] - Aula [Número]'. Exemplo: 'Docker Avançado - Aula 12' ou apenas diga o nome da tecnologia (ex: 'Flutter')."


# --- 5. LOOP DE INTERAÇÃO DO CHAT ---
agente = AgenteDuvidas("Tutor Super IA")

print(f"[{agente.nome}]: Motor carregado com sucesso com {len(documentos)} registros de conhecimento!")
print(" Converse comigo continuamente. Digite sua dúvida ou escreva 'sair' para fechar o chat.\n")

while True:
    pergunta_usuario = input("Aluno: ")
    
    if pergunta_usuario.lower().strip() == "sair":
        print(f"\n[{agente.nome}]: Sessão encerrada com sucesso. Bons estudos!")
        break
        
    resposta = agente.responder(pergunta_usuario)
    print(f"[{agente.nome}]: {resposta}\n")
